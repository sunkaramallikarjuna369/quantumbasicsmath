def kron_all(matrices):
    """
    Compute Kronecker product of a list of matrices.
    Used to build multi-qubit operators like H^{⊗n}.
    """
    result = np.array([[1]], dtype=complex)
    for m in matrices:
        result = np.kron(result, m)
    return result
Cell 33 [Code]
def H_n(n):
    """
    Hadamard on n qubits: H ⊗ H ⊗ ... ⊗ H
    """
    return kron_all([H] * n)
Cell 34 [Code]
def oracle_Uf(n, f):
    """
    Build the oracle matrix Uf for n input qubits + 1 ancilla qubit.

    Basis ordering:
      |x>|y>  → index = (x << 1) | y
    """
    dim = 2 ** (n + 1)
    U = np.zeros((dim, dim), dtype=complex)

    print("\n[Oracle Construction]")
    for x in range(2**n):
        fx = int(f(x)) & 1
        print(f"  f({format(x, f'0{n}b')}) = {fx}")
        for y in (0, 1):
            in_idx  = (x << 1) | y
            out_idx = (x << 1) | (y ^ fx)
            U[out_idx, in_idx] = 1.0

    return U

def deutsch_jozsa(n, f):
    """
    Run Deutsch–Jozsa algorithm for n input qubits.
    Prints intermediate states for learning.
    """

    print("\n" + "=" * 65)
    print(f"Running Deutsch–Jozsa for n = {n}")
    print("=" * 65)

    dim = 2 ** (n + 1)

    # ---------------------------------------------------
    # Step 1: Initialize |0^n>|1>
    # ---------------------------------------------------
    state = np.zeros(dim, dtype=complex)
    state[1] = 1.0  # |x=0>|y=1>

    print("\nStep 1: Initial state |0...0>|1>")
    print(state)

    # ---------------------------------------------------
    # Step 2: Apply H on all qubits
    # Result: |+...+>|->
    # ---------------------------------------------------
    print("\nStep 2: Apply H^{⊗(n+1)}")
    state = H_n(n + 1) @ state
    print("State after Hadamards:")
    print(state)

    # ---------------------------------------------------
    # Step 3: Apply oracle U_f
    # Phase kickback occurs here
    # ---------------------------------------------------
    print("\nStep 3: Apply oracle U_f")
    U = oracle_Uf(n, f)
    state = U @ state
    print("State after oracle:")
    print(state)

    # ---------------------------------------------------
    # Step 4: Apply H^{⊗n} on input register only
    # ---------------------------------------------------
    print("\nStep 4: Apply H^{⊗n} on input qubits")
    H_input = kron_all([H]*n + [I])
    state = H_input @ state
    print("Final state:")
    print(state)

    # ---------------------------------------------------
    # Step 5: Measure input register probabilities
    # ---------------------------------------------------
    print("\nStep 5: Measure input register (ignore ancilla)")
    probs = {}
    for x in range(2**n):
        amp0 = state[(x << 1) | 0]
        amp1 = state[(x << 1) | 1]
        p = abs(amp0)**2 + abs(amp1)**2
        if p > 1e-10:
            probs[format(x, f"0{n}b")] = p

    for k, v in probs.items():
        print(f"  P(|{k}>) = {v:.6f}")

    # ---------------------------------------------------
    # Step 6: Deutsch–Jozsa decision
    # ---------------------------------------------------
    if probs.get("0"*n, 0.0) > 1 - 1e-9:
        decision = "CONSTANT"
    else:
        decision = "BALANCED"

    print("\nDeutsch–Jozsa Decision:", decision)
    return probs, decision
